// ดักจับเมื่อผู้ใช้เลือกเปลี่ยนสิทธิ์อุปกรณ์ function requestRoleChange(newRole) { if (newRole === 'customer') { changeRoleMode('customer'); return; } // เปิดหน้าต่าง Login เพื่อขอรหัสผ่าน targetRoleToSwitch = newRole; document.getElementById('login-pin-input').value = ''; document.getElementById('login-error-msg').classList.add('hidden'); const titles = { cashier: "เข้าสู่ระบบเครื่องแคชเชียร์ (POS)", kds: "เข้าสู่ระบบจอแสดงผลในครัว (KDS)", waiter: "เข้าสู่ระบบพนักงานเสิร์ฟ" }; document.getElementById('login-title').innerText = titles[newRole] || "เข้าสู่ระบบหลังบ้าน"; document.getElementById('login-modal').classList.remove('hidden'); } // ตรวจสอบความถูกต้องของรหัส PIN function handleAdminLogin(e) { e.preventDefault(); const pin = document.getElementById('login-pin-input').value; const requiredPin = ROLE_PASSWORDS[targetRoleToSwitch]; if (pin === requiredPin || pin === ROLE_PASSWORDS.cashier) { document.getElementById('login-modal').classList.add('hidden'); changeRoleMode(targetRoleToSwitch); } else { document.getElementById('login-error-msg').classList.remove('hidden'); } } // กรณีกดปุ่มยกเลิก function cancelRoleChange() { document.getElementById('login-modal').classList.add('hidden'); document.getElementById('role-selector').value = currentRole; }
